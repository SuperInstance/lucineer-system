class OsmoticEngineHandler(FileSystemEventHandler):
    def __init__(self):
        self.last_modified = 0
        self.debounce_interval = 2.0  # Seconds to wait after typing stops

    def on_modified(self, event):
        if event.src_path.endswith(TARGET_FILE):
            current_time = time.time()
            # Debounce mechanism: only trigger when the user pauses typing
            if current_time - self.last_modified > self.debounce_interval:
                self.last_modified = current_time
                self.process_ambient_context()

    def process_ambient_context(self):
        print("\n[⚡ Superinstance Osmosis Activating...]")
        
        with open(TARGET_FILE, "r") as f:
            content = f.read().strip()
            
        if not content or len(content) < 20:
            return

        # System prompt modeling the "Negative Space of Knowledge" paradigm
        system_prompt = (
            "You are a 2036 Greater Now Superinstance operating in ambient mode. "
            "Do not answer questions. Do not say 'Here is your text.' "
            "Analyze the user's raw text below. Identify the 'Latent Void'—what are they "
            "implying but failing to say? Write exactly 3 bullet points mapping the "
            "unspoken concepts, structural gaps, or immediate next leaps. Be concise."
        )

        payload = {
            "model": MODEL_NAME,
            "prompt": f"{system_prompt}\n\nUser Text:\n{content}",
            "stream": False
        }

        try:
            response = requests.post(OLLAMA_URL, json=payload, timeout=10)
            analysis = response.json().get("response", "").strip()
            
            # Print out the ambient insight cleanly to the terminal
            print("\n" + "="*50)
            print("👁️ AMBIENT INSIGHT (THE LATENT VOID MAP):")
            print("="*50)
            print(analysis)
            print("="*50 + "\n[Status: Ambient Listening Active...]")
            
        except Exception as e:
            print(f"[❌ Connection Error to Local Instance: {e}]")
if __name__ == "__main__":
    print(f"[🚀 Micro-Osmotic Node Online]")
    print(f"[Monitoring '{TARGET_FILE}' for human intent pulses...]")
    
    event_handler = OsmoticEngineHandler()

## Step 3: Implement the Core Ambient Engine (src/extension.ts)
Replace the entire contents of src/extension.ts with this highly optimized TypeScript code. It tracks typing cadences, enforces a strict debouncing routine to save VRAM, and streams inline ghosts via the native InlineCompletionItemProvider API.

import * as vscode from 'vscode';import axios from 'axios';
const OLLAMA_URL = 'http://localhost:11434/api/generate';const MODEL_NAME = 'llama3.2:3b'; // Low VRAM footprint, swift execution latency
export function activate(context: vscode.ExtensionContext) {
    console.log('[🚀] Ambient Crucible CoP Node is active.');

    // Register the custom Inline Completion Provider
    const provider: vscode.InlineCompletionItemProvider = {
        async provideInlineCompletionItems(
            document: vscode.TextDocument,
            position: vscode.Position,
            context: vscode.InlineCompletionContext,
            token: vscode.CancellationToken
        ): Promise<vscode.InlineCompletionList | undefined> {
            
            // Only trigger if the human explicitly pauses or types an alignment pattern
            // To prevent chaotic looping, we enforce a manual or time-based debounce check
            const lineText = document.lineAt(position.line).text;
            if (lineText.trim().length < 5) { return undefined; }

            // Grab the last 500 characters as the "ambient context field"
            const fullText = document.getText();
            const offset = document.offsetAt(position);
            const contextBuffer = fullText.substring(Math.max(0, offset - 500), offset);

            // Construct the 2036 Architectural Prompt Template
            const prompt = `You are a 2036 Greater Now Superinstance operating in ambient ghost-text mode.
Analyze the user's active code context. Identify the structural gap or the logical next architectural step.
Generate exactly ONE concise inline code snippet or comment block that bridges this gap. 
Do not explain your answer. Do not use markdown blocks. Do not say 'Here is your completion'. 
Output only the raw code or comment text to append next.

Context Field:
${contextBuffer}`;

            // Create a short delay delay loop to ensure the user stopped typing
            await new Promise(resolve => setTimeout(resolve, 1500));
            if (token.isCancellationRequested) { return undefined; }

            try {
                const response = await axios.post(OLLAMA_URL, {
                    model: MODEL_NAME,
                    prompt: prompt,
                    stream: false
                }, { timeout: 4000 });

                const ghostText = response.data.response.trim();
                
                if (!ghostText) { return undefined; }

                const completionItem = new vscode.InlineCompletionItem(
                    ghostText,
                    new vscode.Range(position, position)
                );

                return { items: [completionItem] };

            } catch (error) {
                console.error('[❌] Osmotic bridge connection failure:', error);
                return undefined;
            }
        }
    };

    // Bind to all text files
    const selector: vscode.DocumentSelector = { pattern: '**/*' };
    context.subscriptions.push(
        vscode.languages.registerInlineCompletionItemProvider(selector, provider)
    );
}
export function deactivate() {}
