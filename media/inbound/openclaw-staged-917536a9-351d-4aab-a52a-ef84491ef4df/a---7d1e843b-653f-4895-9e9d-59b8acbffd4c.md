## The Configuration Core Files
To allow your coder to spin this up instantly with a single command inside a Codespace, hand them these baseline orchestrator files:
## 1. Master Workspace Definition (Cargo.toml)

# /Cargo.toml
[workspace]
members = [
    "quilt_storage",
    "exoj_kernel",
    "eos_runtime"
]

[profile.release]
opt-level = 3
lto = true
codegen-units = 1
panic = "abort"

## 2. Hardware Engine Acceleration Profiles (.cargo/config.toml)

# /.cargo/config.toml
[build]
rustflags = [
    "-C", "target-cpu=native",
    "-C", "link-args=-Wl,-z,stack-size=8388608" # Protects deep search recursion lines
]

## 3. Runtime Component Inter-dependencies (eos_runtime/Cargo.toml)

# /eos_runtime/Cargo.toml
[package]
name = "eos_runtime"
version = "0.1.0"
edition = "2021"

[dependencies]
