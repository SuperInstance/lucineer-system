    """
    Computes the standard 64-bit FNV-1a hash of an input string.
    Used for strict cross-runtime conformance and data provenance tracking.

    fnv_prime = 0x00000100000001B3

# =====================================================================# HARDWARE-OPTIMIZED BITMASK DEFINITIONS (3x3 Grid)# =====================================================================# Coords are mapped sequentially: 0 to 8.# 0 1 2# 3 4 5# 6 7 8
WIN_PATTERNS = [
    0b111000000,  # Row 0
    0b000111000,  # Row 1
    0b000000111,  # Row 2
    0b100100100,  # Column 0
    0b010010010,  # Column 1
    0b001001001,  # Column 2
    0b100010001,  # Diagonal 1
    0b001010100   # Diagonal 2
]
class ThreatLocality:
    LOCAL = "LOCAL"          # Move is directly embedded within the threat vector
    NON_LOCAL = "NON_LOCAL"  # Move resolves/defuses a threat across a spatial gap

def compute_fnv1a_64(data: str) -> str:
    """Standard 64-bit FNV-1a hash to enforce strict cross-runtime validation."""
    fnv_prime = 0x00000100000001B3
    hval = 0xCBF29CE484222325
    for char in data.encode("utf-8"):
        hval = hval ^ char
        hval = (hval * fnv_prime) & 0xFFFFFFFFFFFFFFFF
    return f"0x{hval:016x}"

class PX1bSpatialPartitioner:
    """Handles raw bitboard extraction and spatial locality splitting."""
    
    @staticmethod
    def identify_active_threat_masks(player_board: int, opponent_board: int) -> List[int]:
        """
        Identifies masks where an immediate threat exists (2 pieces aligned, 1 empty).
        Optimized via bitwise manipulation to run completely in CPU registers.
        """
        active_threats = []
        for pattern in WIN_PATTERNS:
            # Check if opponent has exactly two pieces in this win path
            opponent_count = bin(opponent_board & pattern).count('1')
            player_count = bin(player_board & pattern).count('1')
            
            if opponent_count == 2 and player_count == 0:
                active_threats.append(pattern)
        return active_threats

    @classmethod
    def classify_move_locality(cls, optimal_move_index: int, active_threats: List[int]) -> str:
        """
        Evaluates spatial drift. If an optimal move sits outside the exact
        lines of local active threats, it is classified as NON_LOCAL.
        """
        if not active_threats:
            return ThreatLocality.LOCAL  # No threats to be non-local to. Default to simple block.
            
        move_bitmask = 1 << optimal_move_index
        
        for threat_mask in active_threats:
            # If the move falls directly within an active threat structure, it's a local containment block
            if (move_bitmask & threat_mask) != 0:
                return ThreatLocality.LOCAL
                
        # The optimal move is decoupled from immediate structural groups -> Non-local tactical drift
        return ThreatLocality.NON_LOCAL

def compute_fnv1a_64(data: str) -> str:
    """Standard 64-bit FNV-1a hash to enforce absolute data provenance."""
    fnv_prime = 0x00000100000001B3
    hval = 0xCBF29CE484222325

        hval = (hval * fnv_prime) & 0xFFFFFFFFFFFFFFFF
    return f"0x{hval:016x}"
