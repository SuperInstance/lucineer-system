## 1. Mathematical Rigor: The Fixed-Point 16-bit Architecture
To implement this in raw, vendor-agnostic Verilog-2005 without bit-leakage or precision degradation, we define a custom 16-bit Fixed-Point (Q8.8 Format) fractional arithmetic system.
$$\text{Value} = \frac{\text{Integer Internal Representation}}{2^8}$$ 

* Bits [15:8]: Signed Integer component ($-128$ to $+127$).
* Bits [7:0]: Fractional component (Precision of $\frac{1}{256} \approx 0.00390625$).

## The Prevention of Truncation Biases
Standard right-shifting (>>> 8) drops the remaining bits, introducing a negative truncation bias that will slowly destroy the invariant relationship $\gamma + \eta = C$ over thousands of consecutive iterations.
To eliminate this bias, we must implement Convergent Rounding (Round-to-Nearest-Even / Symmetric Rounding) directly in the RTL combinational wiring:

   1. Evaluate the dropped bits.
   2. If the fractional remainder is greater than half the least significant bit (LSB), round up.
   3. If it is exactly half, round to the nearest even number.

module quilt_fused_compute (
    input  wire        clk,
    input  wire        rst_n,
    
    // Ingress Interface
    input  wire [15:0] cell_a_gamma,   // Q8.8 fixed-point
    input  wire [15:0] cell_b_eta,     // Q8.8 fixed-point
    input  wire [15:0] scale_factor,   // Q8.8 fixed-point scaling factor
    input  wire        ingress_valid,
    
    // Egress Interface
    output reg  [15:0] fused_gamma_out, // Processed and rounded output
    output reg  [15:0] fused_eta_out,   // Dynamic counterpart output
    output reg         egress_valid,
    
    // Invariant Verification Hook (For Formal Proving)
    output wire [31:0] debug_total_capacity
);

    // Target Invariant Constant C in Q16.16 Format (For wide calculation range)
    // Assuming constant C = 256.0000 -> 32'h01000000
    localparam [31:0] TARGET_CONSTANT_C_Q16 = 32'h01000000;

    // ------------------------------------------------------------------------
    // Pipeline Stage 1 Registers: Fused Aggregation
    // ------------------------------------------------------------------------
    reg signed [16:0] p1_sum; // 17-bit to prevent overflow on immediate addition
    reg signed [15:0] p1_scale;
    reg               p1_valid;

    // ------------------------------------------------------------------------
    // Pipeline Stage 2 Registers: High-Width Multiplication
    // ------------------------------------------------------------------------
    reg signed [32:0] p2_product; // 33-bit to hold 17-bit * 16-bit signed multiplication
    reg               p2_valid;

    // ------------------------------------------------------------------------
    // Pipeline Architecture
    // ------------------------------------------------------------------------
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            p1_sum          <= 17'sh0;
            p1_scale        <= 16'sh0;
            p1_valid        <= 1'b0;
            
            p2_product      <= 33'sh0;
            p2_valid        <= 1'b0;
            
            fused_gamma_out <= 16'h0;
            fused_eta_out   <= 16'h0;
            egress_valid    <= 1'b0;
        end else begin
            
            // ----------------------------------------------------------------
            // STAGE 1: Immediate Cell Fusion
            // ----------------------------------------------------------------
            if (ingress_valid) begin
                p1_sum   <= $signed({1'b0, cell_a_gamma}) + $signed({1'b0, cell_b_eta});
                p1_scale <= $signed(scale_factor);
                p1_valid <= 1'b1;
            end else begin
                p1_valid <= 1'b0;
            end

            // ----------------------------------------------------------------
            // STAGE 2: Register-Persistent Multiplication
            // ----------------------------------------------------------------
            if (p1_valid) begin
                p2_product <= p1_sum * p1_scale;
                p2_valid   <= 1'b1;
            end else begin
                p2_valid   <= 1'b0;
            end

            // ----------------------------------------------------------------
            // STAGE 3: Convergent Rounding and Mass Invariant Reconstruction
            // ----------------------------------------------------------------
            if (p2_valid) begin
                // Compute Convergent Rounding (Round to Nearest Even) for Q8.8
                // We are dropping the lower 8 bits (p2_product[7:0])
                if ((p2_product[7] == 1'b1) && 
                    ((p2_product[6:0] != 7'b0) || (p2_product[8] == 1'b1))) begin
                    // Round up if remainder > 0.5 OR (remainder == 0.5 AND LSB is odd)
                    fused_gamma_out <= p2_product[23:8] + 16'sh1;
                end else begin
                    fused_gamma_out <= p2_product[23:8];
                end

                // Derivation of the counterpart via the explicit conservation law:
                // eta = C - gamma
                // This ensures zero bit leakage across iterations.
                fused_eta_out <= TARGET_CONSTANT_C_Q16[23:8] - (p2_product[23:8]);
                
                egress_valid  <= 1'b1;
            end else begin
                egress_valid  <= 1'b0;
            end
        end
    end

    // Continuous assignment mapping for structural monitoring hooks
    assign debug_total_capacity = {16'h0, fused_gamma_out} + {16'h0, fused_eta_out};
endmodule

// ============================================================================// Module: quilt_instruction_decoder// Language: Pure Synthesizable Verilog-2005 (IEEE 1364-2005)// Description: Opcode-driven cellular execution unit for the Quilt Calculus.// ============================================================================
module quilt_instruction_decoder (
    input wire         clk,
    input wire         rst_n,

    // Instruction Ingress Port
    input wire [63:0]  i_instruction_word,
    input wire         i_instr_valid,
    output reg         o_decoder_ready,

    // Core Runtime Registration Matrix Interface
    output reg [15:0]  o_target_addr,
    output reg [15:0]  o_linked_addr,
    output reg [15:0]  o_gamma_payload,
    output reg [15:0]  o_eta_payload,
    
    // Decoded Trigger Control Strobe Lines
    output reg         o_strobe_bind,
    output reg         o_strobe_link,
    output reg         o_strobe_effect,
    output reg         o_strobe_view,
    output reg         o_strobe_tick
);

    // Opcode Extraction Aliases
    localparam [2:0] OP_NOP    = 3'b000;
    localparam [2:0] OP_BIND   = 3'b001;
    localparam [2:0] OP_LINK   = 3'b010;
    localparam [2:0] OP_EFFECT = 3'b011;
    localparam [2:0] OP_VIEW   = 3'b100;
    localparam [2:0] OP_TICK   = 3'b101;

    // Single-cycle instruction parse pipeline
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            o_target_addr    <= 16'h0;
            o_linked_addr    <= 16'h0;
            o_gamma_payload  <= 16'h0;
            o_eta_payload    <= 16'h0;
            
            o_strobe_bind    <= 1'b0;
            o_strobe_link    <= 1'b0;
            o_strobe_effect  <= 1'b0;
            o_strobe_view    <= 1'b0;
            o_strobe_tick    <= 1'b0;
            o_decoder_ready  <= 1'b1;
        end else begin
            if (i_instr_valid && o_decoder_ready) begin
                // Synchronously extract data fields
                o_target_addr   <= i_instruction_word[60:45];
                o_linked_addr   <= i_instruction_word[44:29];
                o_gamma_payload <= i_instruction_word[28:13];
                o_eta_payload   <= i_instruction_word[12:0];

                // Clear all prior control strobe signals to avoid latch behaviors
                o_strobe_bind   <= 1'b0;
                o_strobe_link   <= 1'b0;
                o_strobe_effect <= 1'b0;
                o_strobe_view   <= 1'b0;
                o_strobe_tick   <= 1'b0;

                // Decode and strobe the matching operational unit
                case (i_instruction_word[63:61])
                    OP_BIND:   o_strobe_bind   <= 1'b1;
                    OP_LINK:   o_strobe_link   <= 1'b1;
                    OP_EFFECT: o_strobe_effect <= 1'b1;
                    OP_VIEW:   o_strobe_view   <= 1'b1;
                    OP_TICK:   o_strobe_tick   <= 1'b1;
                    default:   /* NOP or Malformed Execution Frame */;
                endcase
                
                o_decoder_ready <= 1'b0; // Lock decoder while downstream loops catch up
            end else begin
                // Automatic automatic release of structural handshakes
                o_strobe_bind   <= 1'b0;
                o_strobe_link   <= 1'b0;
                o_strobe_effect <= 1'b0;
                o_strobe_view   <= 1'b0;
                o_strobe_tick   <= 1'b0;
                o_decoder_ready <= 1'b1;
            end
        end
    end
endmodule

## 2. The Solution: Convergent Rounding (Round-to-Nearest-Even)
To eliminate mathematical drift, your arithmetic unit must implement Convergent Rounding (also known as Statistician's Rounding) directly in the combinational wiring.
When truncating a value, the hardware inspects the bits being dropped to determine how to round the result:

* If the dropped bits are greater than half (> 4'b1000), round up ($+1$).
* If the dropped bits are less than half (< 4'b1000), round down ($+0$).
* If the dropped bits are exactly half (= 4'b1000), round to whichever option makes the least significant bit (LSB) of the remaining number even (0).

## The Convergent Logic Function
Let $R$ be the remaining upper bit-field, $D$ be the highest dropped bit (the MSB of the discarded fraction), and $T$ be the logical OR of all subsequent lower dropped bits. We can express the rounding decision as a boolean equation:
$$\text{Round Up Strobe} = D \ \text{AND} \ (T \ \text{OR} \ R_0)$$ 
Where $R_0$ is the current LSB of your remaining value. This choice balances the rounding errors perfectly. Statistically, half of the operations round up and half round down, reducing the cumulative mathematical drift to zero.

   Y Axis

     │    /
     │   /  Rotation Steps (Shifts & Adds)

module quilt_cordic_engine (
    input wire         clk,
    input wire         rst_n,
    input wire         i_vector_valid,
    input wire [15:0]  i_x,              // Signed Q8.8 coordinate input
    input wire [15:0]  i_y,              // Signed Q8.8 coordinate input

    output reg         o_magnitude_valid

    // ------------------------------------------------------------------------

    reg               s2_valid;

    reg               s3_valid;

            s2_x <= 16'sh0; s2_y <= 16'sh0; s2_valid <= 1'b0;
            s3_x <= 16'sh0; s3_y <= 16'sh0; s3_valid <= 1'b0;

                s1_y     <= (i_y[15] == 1'b1) ? -i_y : i_y;
                s1_valid <= 1'b1;
            end else begin
                s1_valid <= 1'b0;
            end

            // STAGE 2: Core Iteration 1 (Shift right by 0 bits)
            if (s1_valid) begin
                s2_valid <= 1'b1;

                    s2_y <= s1_y - s1_x;
                end else begin              // Y is negative, rotate counter-clockwise

                s2_valid <= 1'b0;

                    s3_y <= s2_y - (s2_x >>> 1);
                end else begin

                    s3_y <= s2_y + (s2_x >>> 1);

            end
        end
    end
